#+begin_src emacs-lisp

    (interactive)
    (find-file "~/.emacs.d/base_cfg.org"))
  (global-set-key (kbd "C-c 9") 'config-visit)

  (defun config-reload ()
    (interactive)
    (org-babel-load-file (expand-file-name "~/.emacs.d/base_cfg.el")))
  (global-set-key (kbd "C-c 0") 'config-reload)
